        }
\end{proc}

\begin{proc}{registrarUsuario}
{
\Inout am:AirMiles, 
\In usuario:Usuario
}
{}
    \requiere{am=AM_0   \land  usuario \notin AM_0.historiales }
    \aseguraLargo{am.historiales = setKey(AM_0.historiales,usuario,<canjeadosreferidos:<>,vuelos:<>,transferencias:<>\ >) \ \land \
        am.promocionesSinTope = AM_0.promocionesSinTope \land \\am.promocionesConTope = AM_0.promocionesConTope
        }
\end{proc}

\begin{proc}{registrarUsuarioConReferidor}
{
\Inout am: AirMiles, 
\In usuarioReferido: Usuario, \\
\In usuarioReferidor: Usuario
}
{} 
    \requiere{am=AM_0  \land usuarioReferidor \in AM_0.historiales \land\\
    usuarioReferido \notin AM_0.historiales} 
    \aseguraLargo{am.promocionesSinTope = AM_0.promocionesSinTope \land \\ am.promocionesConTope = AM_0.promocionesConTope \land \\
    am.historiales[usuarioReferido] = <canjeadosreferidos: <("Referido",500)> , \\ vuelos:<>,transferencias:<>>\land \\
    esCanjeadosReferidosActualizado(am.historiales[usuarioReferidor],\\AM_0.historiales[usuarioReferidor],"Referidor",500)\land \\
    igualSalvo(am.historiales,AM_0.historiales,\tupla{usuarioReferido,usuarioReferidor})
                    }
\end{proc}


\begin{proc}{acumularMillas}
{
\Inout am:AirMiles,
\In usuario:Usuario,
\In distancia: Distancia,
\In promocion: NombrePromocion
}{}
    \requiere{am = AM_0 \ \land \ usuario \in AM_0.historiales \ \land \ (promocion \in AM_0.promocionesSinTope \ \lor promocion \in AM_0.promocionesConTope) \land_L \\
                (esPromocionSinTope(AM_0.promociones, promocion) \ \lor_L \\ noSePasaDeTope(AM_0, usuario, distancia, promocion) )               
    }
    \aseguraLargo{  esVueloActualizado(am,AM_0,usuario,distancia,categoria(AM_0.historial[usuario]),promocion)\ \land am.promocionesSinTope = AM_0.promocionesSinTope \land \\
    (esPromocionSinTope(AM_0, promocion) \implies am.promocionesConTope = AM_0.promocionesConTope) \\ \land \\ (esPromocionConTope(AM_0, promocion) \ \land \ alcanzaTope(AM_0, distancia,\\ categoria(AM_0.historial[usuario]), promocion) \implies \\ am.promocionesConTope = delKey(AM_0.promocionesConTope, promocion)) \\ \land \\ (esPromocionConTope(AM_0, promocion) \ \land \\ \lnot alcanzaTope(AM_0, distancia, categoria(AM_0.historial[usuario]), promocion) \implies \\ restaDelTope(am, AM_0, distancia, categoria(AM_0.historial[usuario]), promocion)))) \\
          }

\end{proc}


\begin{proc}{canjearMillas}
{
\Inout am:AirMiles,
\In usuario:Usuario,
\In millas:Millas
}{}
    \requiere{am=AM_0\  \land \ usuario\in AM_0.historiales  \land \\ millas \leq millasDisponibles(AM_0.historiales[usuario])}
    \aseguraLargo{
        am.promocionesSinTope = AM_0.promocionesSinTope \land \\
        am.promocionesConTope = AM_0.promocionesConTope \land \\
        igualSalvo(am.historiales,AM_0.historiales,\tupla{usuario}) \land \\
        esCanjeadosReferidosActualizado(am.historiales[usuario],\\AM_0.historiales[usuario],"Canjear",millas) 
                }
\end{proc}


\begin{proc}{transferirMillas}
{
\Inout am:AirMiles,
\In dador : Usuario,
\In receptor : Usuario,\\
\In millas : Millas
}{}    
    \requiere{am=AM_0 \land dador \in AM_0.historiales \land \\receptor \in AM_0.historiales \land \\ dador \neq receptor \land \\ millasDisponibles(AM_0.historiales[dador]) \geq millas}
    \aseguraLargo{am.promocionesSinTope = AM_0.promocionesSinTope \land \\
        am.promocionesConTope = AM_0.promocionesConTope \land \\
                igualSalvo(am.historiales,AM_0.historiales,\tupla{dador,receptor}) \land \\
                esTransferenciasActualizado(am.historiales[dador],\\AM_0.historiales[dador],"Transferir",millas,receptor) \land \\
                esTransferenciasActualizado(am.historiales[receptor],\\AM_0.historiales[receptor],"Recibir",millas,dador) 
                    }  
\end{proc}

\begin{proc}{consultarSaldo}
{
\In am:AirMiles,
\In usuario:Usuario
}{
\Tipo{Millas}
}
    \requiere{usuario \in am.historiales}
    \aseguraLargo{res = millasDisponibles(am.historiales[usuario])}
\end{proc}

\begin{proc}{consultarCategoria}{
\In am:AirMiles,
\In usuario:Usuario
}{
\Tipo{Categoria}
}
    \requiere{usuario \in am.historiales}
    \aseguraLargo{(res = "Bronce" \iff esBronce(am.historiales[usuario]) \land \\
            (res = "Plata" \iff  esPlata(am.historiales[usuario]) \land \\
            (res = "Oro" \iff esOro(am.historiales[usuario]) \land \\
            (res = "Platino" \iff esPlatino(am.historiales[usuario]) \\
            }
\end{proc}

\begin{proc}{rankingMillas}
{
\In am:AirMiles,
\In n:$\mathbb{Z}$
}{
\Tipo{\seq{Usuario}}
}
    \requiere{|am.historiales|>0\land n > 0}
    \aseguraLargo{
        (|am.historiales|<n \implies |res| = |am.historiales|) \land \\
        (|am.historiales|\geq n \implies |res| = n) \land \\ 
        sinRepetidos(res) \land \\
        ((\forall u \in Usuario)  \ u\in res \implies u \in am.historiales) \land_L \\
        listaOrdenadaPorMillasDisponibles(am, res) \land \\\lnot existeUsuarioFueraDeListaConMasMillasQueElQueMenosTiene(am, res)                       
    }
\end{proc}





\begin{proc}{resumenTransacciones}
{
\In am:AirMiles,
\In usuario:Usuario
}{ \\
\Tipo{\tupla{\tupla{Accion,Millas},\tupla{Accion,Millas},\tupla{Accion,Millas},\tupla{Accion,Millas}}}
}
    \requiere{usuario \in am.historiales}
    \aseguraLargo{res_0 = \tupla{"Acumular",acumuladoPorViajes(am.historiales[usuario])} \land \\

                res_3= \tupla{"Recibir",totalRecibidas(am.historiales[usuario])}}
\end{proc}


\begin{proc}{iniciarPromocion}
{
\Inout am:AirMiles,
\In nombrePromocion : NombrePromocion,
\In factor : Factor
}{}
    \requiere{am=AM_0 \land factor \geq 1 \land nombrePromocion \notin AM_0.promocionesSinTope \land nombrePromocion \notin AM_0.promocionesConTope}
    \aseguraLargo{am.historiales = AM_0.historiales \land \\
        am.promocionesConTope = AM_0.promocionesConTope \land \\
        am.promocionesSinTope = setKey(AM_0.promocionesSinTope,nombrePromocion, factor)       
        }
\end{proc}


\begin{proc}{iniciarPromocionConTope}
{
\Inout am:AirMiles,
\In nombrePromocion : NombrePromocion, \\
\In factor : Factor,
\In tope : montoTope
}{}
    \requiere{am=AM_0 \land factor \geq 1 \land nombrePromocion \notin AM_0.promocionesSinTope \land nombrePromocion \notin AM_0.promocionesConTope}
    \aseguraLargo{am.historiales = AM_0.historiales \land \\
        am.promocionesSinTope = AM_0.promocionesSinTope \land \\
        am.promocionesConTope = setKey(AM_0.promocionesConTope,nombrePromocion, \struct{factor:factor,tope:tope})
    }
\end{proc}


\begin{proc}{finalizarPromocion}
{
\Inout am:AirMiles,
\In nombrePromocion : NombrePromocion
}{}
    \requiere{am=AM_0\ \land  \ (nombrePromocion \in AM_0.promocionesSinTope \ \lor \\ nombrePromocion \in AM_0.promocionesConTope) \land \ nombrePromocion \neq "NoPromo"}
    \aseguraLargo{am.historiales = AM_0.historiales \land \\
        (esPromocionSinTope(AM_0,nombrePromocion) \implicaLuego \\(am.promocionesSinTope = delKey(AM_0.promocionesSinTope,nombrePromocion)\ \land \\ am.promocionesConTope = AM_0.promocionesConTope)) \\ \land \\ (esPromocionConTope(AM_0,nombrePromocion) \implicaLuego \\(am.promocionesConTope = delKey(AM_0.promocionesConTope,nombrePromocion)\ \land \\ am.promocionesSinTope = AM_0.promocionesSinTope))
    }
\end{proc}

\begin{proc}{paresTransferidoresExclusivos}
{
\In am:AirMiles
}{
\Tipo{\seq{tupla\tupla{Usuario,Usuario}}}
}
    \requiere{True}
    \aseguraLargo{
        sinRepetidos(res) \land sinAnagramas(res) \land \\
        (\forall i,j \in Z)(((i,j \in am.historiales \ \land i\neq j)\ \yLuego \\ sonParejaTransferidores(am.historiales,i,j))\implicaLuego (\tupla{i,j} \in res \lor \tupla{j,i} \in res )) \land \\ 
        (\forall k : Z)(0\leq k < |res| \implicaLuego ((res[k]_0 \in am.historiales \land res[k]_1\in am.historiales) \ \yLuego \\ sonParejaTransferidores(am.historiales,res[k]_0,res[k]_1)))
    }
\end{proc}

\pred{esCanjeadosReferidosActualizado}{hist:Historial,hist0:Historial,accion:Accion,millas:Millas}{
hist = <canjeadosreferidos : hist0.canjeadosreferidos++\tupla{accion,millas}, vuelos : hist0.vuelos, transferencias : hist0.transferencias  >
}
\\

\pred{esTransferenciasActualizado}{hist:Historial,hist0:Historial,accion:Accion,millas:Milla,usuario:Usuario}{\\
hist = <canjeadosreferidos : hist0.canjeadosreferidos, vuelos : hist0.vuelos, transferencias : hist0.transferencias++\tupla{accion,millas,usuario}  >
}

\pred{igualSalvo}{dict1: dicc<K, V>, dict2: dicc<K, V>, ignorar: \conj{Usuario}}{\\
    (\forall i : K) ((i \in dict1 \land i \notin ignorar) \implicaLuego (i \in dict2 \yLuego dict1[i] = dict2[i])) \land \\
    (\forall j : K) (j \in dict2 \land j \notin ignorar) \implicaLuego (j \in dict1 \yLuego dict1[j] = dict2[j]) 
}

\\
\\
\pred{esPromocionConTope}{am : AirMiles, promocion : NombrePromocion}{ \\promocion \in am.promocionesConTope}

\pred{esVueloActualizado}{am : AirMiles, AM_0 : AirMiles, u : Usuario, d : Distancia, cat : R, p : NombrePromocion}{ \\
(esPromocionConTope(AM_0,p) \implicaLuego (am.historiales = setKey(AM_0.historiales, u,   <canjeadosreferidos: AM_0.historiales[u].canjeadosreferidos, vuelos: AM_0.historiales[u].vuelos ++ \\\seq{\tupla{d,d*AM_0.promocionesConTope[p].factor*cat,p,AM_0.promocionesConTope[p].factor}},\\ transferencias: AM_0.historiales[u].transferencias>))) \\ \land \\
(esPromocionSinTope(AM_0,p) \implicaLuego (am.historiales = setKey(AM_0.historiales, u,   <canjeadosreferidos: AM_0.historiales[u].canjeadosreferidos, vuelos: AM_0.historiales[u].vuelos ++ \\\seq{\tupla{d,d*AM_0.promocionesSinTope[p]*cat,p,AM_0.promocionesSinTope[p]}},\\ transferencias: AM_0.historiales[u].transferencias>)))
}

\pred{sonParejaTransferidores}{hist: Historiales, usuario1:Usuario,\\ usuario2:Usuario}{\\
(|hist[usuario1].transferencias| > 0) \land (|hist[usuario2].transferencias| > 0)  \land \\
(\forall i \in Z)(0 \leq i < |hist[usuario1].transferencias| \implicaLuego (hist[usuario1].transferencias[i]_2 = usuario2) \ \land 
(\forall j \in Z)(0 \leq j < |hist[usuario2].transferencias| \implicaLuego (hist[usuario2].transferencias[j]_2 = usuario1) \\
}

\pred{sinAnagramas}{s : \seq{\tupla{\Z,\Z}}}{
(\forall i \in \Z)(0 \leq i < |s| \implicaLuego noEstaSuAnagrama(s,s[i]))
}
\\
\\
\pred{noEstaSuAnagrama}{s : \seq{\tupla{\Z,\Z}},tupla : \tupla{\Z,\Z}}{\\
\lnot (\exists i \in \Z)(0 \leq i < |s| \implicaLuego (s[i]_0 = tupla_1 \land s[i]_1 = tupla_0))
}
\\

\pred{noSePasaDeTope}{am : AirMiles, usuario : Usuario, distancia : Distancia, \\ promocion : NombrePromocion}{ \\
cantidadMillasAGanar(distancia,am.promocionesConTope[promocion].factor,\\categoria(am.historiales[usuario]) \leq am.promocionesConTope[promocion].tope
}
\\

\pred{alcanzaTope}{am : AirMiles, distancia : Distancia, categoria : R, promocion : NombrePromocion}{am.promocionesConTope[promocion].tope = distancia * categoria * am.promocionesConTope[promocion].factor}
\\
\\
\pred{restaDelTope}{am : AirMiles, AM_0 : AirMiles, distancia : Distancia, categoria : R, promocion : NombrePromocion}{am.promocionesConTope = setKey(AM_0, promocion, \\ \struct{factor: AM_0.promociones[promocion].factor, tope:  AM_0.promociones[promocion].tope- \\cantidadMillasAGanar(distancia,AM_0.promociones[promocion].factor,categoria))}}
\\
\\
\pred{existeUsuarioFueraDeListaConMasMillasQueElQueMenosTiene}{am : AirMiles, s : \seq{Usuario}}{
(\exists u \in Usuario) u \in am.historiales \land  u \notin s \land_L \\millasDisponibles(am.historiales[u]) > millasDisponibles(am.historiales[s[|s|-1]])
}
\\
\\
\pred{listaOrdenadaPorMillasDisponibles}{am : AirMiles, s : \seq{Usuario}}{ \\
(\forall i \in Usuario) \ 0 \leq i < |s| - 1 \implicaLuego millasDisponibles(am.historiales[s[i]]) \geq \\millasDisponibles(am.historiales[s[i+1]])
}
\\

\pred{sinRepetidos}{s : \seq{T}}{
(\forall i \in \Z)(0 \leq i < |s| \implicaLuego cantidadDeApariciones(s,s[i]) = 1) 
}
\\

\aux{categoria}{hist:Historial}{\R}{
IfThenElse(esBronce(hist),1,0) + IfThenElse(esPlata(hist),1.5,0) \\
+ IfThenElse(esOro(hist),2,0)
}
\\
\\
\aux{millasTotales}{hist: Historial}{\R}{ \\ 
\sum_{i=0}^{|hist.vuelos|-1}hist.vuelos[i]_1 + \\
\sum_{i=0}^{|hist.canjeadosreferidos|-1} IfThenElse(hist.canjeadosreferidos[i]_0 = "Canjear" ,\\0,hist.canjeadosreferidos[i]_1)+\\
\sum_{i=0}^{|hist.transferencias|-1}
IfThenElse(hist.transferencias[i]_0 = "Transferir", \\-1*hist.transferencias[i]_1,hist.transferencias[i]_1)
}
\\
\\
\aux{cantidadMillasAGanar}{dist:Distancia,fact:Factor,cate:R}{Millas}{dist*fact*cate}
\\
\\
\aux{millasDisponibles}{hist: Historial}{\R}{\\
\sum_{i=0}^{|hist.vuelos|-1}hist.vuelos[i]_1 + \\
\sum_{i=0}^{|hist.canjeadosreferidos|-1} IfThenElse(hist.canjeadosreferidos[i]_0 = "Canjear" ,\\-1*hist.canjeadosreferidos[i]_1,hist.canjeadosreferidos[i]_1)+\\
\sum_{i=0}^{|hist.transferencias|-1}
IfThenElse(hist.transferencias[i]_0 = "Transferir", \\-1*hist.transferencias[i]_1,hist.transferencias[i]_1)
}
\\
\\
\aux{acumuladoPorViajes}{hist:Historial}{Millas}{ \\ $\sum_{i=0}^{|hist.vuelos|-1}$ hist.vuelo_1}
\\
\\
\aux{totalCanjeadas}{hist:Historial}{Millas}{\\$\sum_{i=0}^{|hist.canjeadosreferidos|-1}$ \\ifThenElse(hist.canjeadosreferidos[i]_0  ="Canjear",hist.canjeadosreferidos[i]_1,0)}
\\
\\
\aux{totalGastadas}{trans:Transacciones}{Millas}{\\$\sum_{i=0}^{|trans|-1}$ \\ifThenElse(trans[i]_0 \in \seq{"Canjear","Transferir"},trans[i]_1,0)}
\\
\\
\aux{totalTransferidas}{hist:Historial}{Millas}{\\$\sum_{i=0}^{|hist.transferencias|-1}$ \\ifThenElse(hist.transferencias[i]_0 = "Transferir",hist.transferencias[i]_1,0)}
\\
\\
\aux{totalRecibidas}{hist:Historial}{Millas}{\\$\sum_{i=0}^{|hist.transferencias|-1}$ \\ifThenElse(hist.transferencias[i]_0 = "Recibir",hist.transferencias[i]_1,0)}
\\
\\
\aux{cantidadDeApariciones}{s : \seq{T}, elem : T}{\Z}{
$\sum_{i=0}^{|s|-1}$ ifThenElse(s[i] = elem,1,0)
\\
