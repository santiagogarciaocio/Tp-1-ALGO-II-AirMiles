    \requiere{True}

        res.promociones =  <"NoPromo":(1,False,0)>  \land \\

        }
\end{proc}

\begin{proc}{registrarUsuario}
{
\Inout am:AirMiles, 
\In usuario:Usuario
}
{}
    \requiere{am=AM_0   \land  usuario \notin AM_0.historiales }
    \aseguraLargo{am.historiales = setKey(AM_0.historiales,usuario,<>) \ \land \
        am.promociones = AM_0.promociones
        }
\end{proc}

\begin{proc}{registrarUsuarioConReferidor}
{
\Inout am: AirMiles, 
\In usuarioReferido: Usuario, \\
\In usuarioReferidor: Usuario
}
{} 
    \requiere{am=AM_0  \land usuarioReferidor \in AM_0.historiales \land\\
    usuarioReferido \notin AM_0.historiales} 
    \aseguraLargo{am.promociones = AM_0.promociones \land \\
    am.historiales[usuarioReferido].transacciones = <\tupla{"Referido",500}> \land \\
    am.historiales[usuarioReferido].vuelos = <> \land \\
    am.historiales[usuarioReferido].transferencias = <> \land \\
    am.historiales[usuarioReferidor].transacciones = \\ AM_0.historiales[usuarioReferidor].transacciones ++ \seq{\tupla{"Referidor",500}} \land \\
    am.historiales[usuarioReferidor].vuelos = AM_0.historiales[usuarioReferidor].vuelos \land \\
    am.historiales[usuarioReferidor].transferencias = \\ AM_0.historiales[usuarioReferidor].transferencias \land \\
    igualSalvo(am.historiales,AM_0.historiales,\tupla{usuarioReferido,usuarioReferidor})
                    }
\end{proc}


\begin{proc}{acumularMillas}
{
\Inout am:AirMiles,
\In usuario:Usuario,
\In distancia: Distancia,
\In promocion: NombrePromocion
}{}
    \requiere{am = AM_0 \land usuario \in AM_0.historiales \land promocion \in AM_0.promociones \land \\
                (promocionSinTope(AM_0.promociones, promocion)\lor \\ noSePasaDeTope(AM_0, usuario, distancia, promocion) )
                
    }
    \aseguraLargo{(esBronce(AM_0.historiales[usuario].transacciones) \implies \\ (historialActualizado(am, AM_0, usuario, distancia, 1, promocion) \land \\ (esPromocionSinTope(AM_0, promocion) \implies am.promociones = AM_0.promociones) \land (esPromocionConTope(AM_0, promocion) \ \land \ alcanzaTope(AM_0, distancia, 1, promocion) \implies am.promociones = delKey(AM_0.promociones, promocion)) \ \land \\ (esPromocionConTope(AM_0, promocion) \ \land \ \lnot alcanzaTope(AM_0, distancia, 1, promocion) \implies restaDelTope(am, AM_0, distancia, 1, promocion)))) \\ \land \\ (esPlata(AM_0.historiales[usuario].transacciones) \implies \\ (historialActualizado(am, AM_0, usuario, distancia, 1.5, promocion) \land \\ (esPromocionSinTope(AM_0, promocion) \implies am.promociones = AM_0.promociones) \land (esPromocionConTope(AM_0, promocion) \ \land \ alcanzaTope(AM_0, distancia, 1.5, promocion) \implies am.promociones = delKey(AM_0.promociones, promocion)) \ \land \\ (esPromocionConTope(AM_0, promocion) \ \land \ \lnot alcanzaTope(AM_0, distancia, 1.5, promocion) \implies restaDelTope(am, AM_0, distancia, 1.5, promocion)))) \\ \land \\ (esOro(AM_0.historiales[usuario].transacciones) \implies \\ (historialActualizado(am, AM_0, usuario, distancia, 2, promocion) \land \\ (esPromocionSinTope(AM_0, promocion) \implies am.promociones = AM_0.promociones) \land (esPromocionConTope(AM_0, promocion) \ \land \ alcanzaTope(AM_0, distancia, 2, promocion) \implies am.promociones = delKey(AM_0.promociones, promocion)) \ \land \\ (esPromocionConTope(AM_0, promocion) \ \land \ \lnot alcanzaTope(AM_0, distancia, 2, promocion) \implies restaDelTope(am, AM_0, distancia, 2, promocion)))) \\ \land \\ 
    (esPlatino(AM_0.historiales[usuario].transacciones) \implies (am.historiales = \\ setKey(AM_0.historiales, usuario, struct<AM_0.historiales[usuario].transacciones, \\ AM_0.historiales[usuario].vuelos ++ \seq{\tupla{distancia,0,promocion,factorPromo(AM_0, promocion)}},\\ AM_0,historiales[usuario].transferencias>) \land am.promociones = AM_0.promociones))}                  

\end{proc}


\begin{proc}{canjearMillas}
{
\Inout am:AirMiles,
\In usuario:Usuario,
\In millas:Millas
}{}
    \requiere{am=AM_0\  \land \ usuario\in AM_0.historiales  \land \\ millas \leq millasDisponibles(AM_0.historiales[usuario].transacciones)}
    \aseguraLargo{
        am.promociones = AM_0.promociones \land \\
        igualSalvo(am.historiales,AM_0.historiales,\tupla{usuario}) \land \\
        am.historial[usuario].transacciones = \\ AM_0.historial[usuario].transacciones ++ \seq{\tupla{"Canjear",millas}} \land \\
        am.historial[usuario].viajes = AM_0.historial[usuario].viajes \land \\
        am.historial[usuario].transferencias = AM_0.historial[usuario].transferencias        
                }
\end{proc}


\begin{proc}{transferirMillas}
{
\Inout am:AirMiles,
\In dador : Usuario,
\In receptor : Usuario,\\
\In millas : Millas
}{}    
    \requiere{am=AM_0 \land dador \in AM_0.historiales \land \\receptor \in AM_0.historiales \land \\ dador \neq receptor \land \\ millasDisponibles(AM_0.historiales[dador].transacciones) \geq millas}
    \aseguraLargo{am.promociones = AM_0.promociones \land \\
                igualSalvo(am.historiales,AM_0.historiales,\tupla{dador,receptor}) \land \\
                am.historial[dador].vuelos = AM_0.historial[dador].vuelos \land \\
                am.historial[dador].transacciones = \\ AM_0.historial[dador].transacciones ++ \seq{\tupla{"Transferir",millas}} \land \\
                am.historial[dador].transferencias = \\ AM_0.historial[dador].transferencias ++ \seq{\tupla{"Transferir",millas,receptor}} \land \\
                am.historial[receptor].vuelos = AM_0.historial[receptor].vuelos \land \\
                am.historial[receptor].transacciones = \\ AM_0.historial[receptor].transacciones ++ \seq{\tupla{"Recibir",millas}} \land \\
                am.historial[receptor].transferencias = \\ AM_0.historial[receptor].transferencias ++ \seq{\tupla{"Recibir",millas,dador}}
                
                    }  
\end{proc}


\begin{proc}{consultarSaldo}
{
\In am:AirMiles,
\In usuario:Usuario
}{
\Tipo{Millas}
}
    \requiere{usuario \in am.historiales}
    \aseguraLargo{res = millasDisponibles(am.historiales[usuario].transacciones)}
\end{proc}

\begin{proc}{consultarCategoria}{
\In am:AirMiles,
\In usuario:Usuario
}{
\Tipo{Categoria}
}
    \requiere{usuario \in am.historiales}
    \aseguraLargo{(res = "Bronce" \iff millasTotales(am.historiales[usuario].transacciones) < 10.000) \land \\
            (res = "Plata" \iff millasTotales(am.historiales[usuario].transacciones) \geq 10.000 \land millasTotales(am.historiales[usuario].transacciones) < 50.000) \land \\
            (res = "Oro" \iff millasTotales(am.historiales[usuario].transacciones) \geq 50.000 \ \land \\ millasTotales(am.historiales[usuario].transacciones) < 1.000.000) \land \\
            (res = "Platino" \iff millasTotales(am.historiales[usuario].transacciones) \geq 1.000.000) \\
            }
\end{proc}

\begin{proc}{rankingMillas}
{
\In am:AirMiles,
\In n:$\mathbb{Z}$
}{
\Tipo{\seq{Usuario}}
}
    \requiere{n > 0}
    \aseguraLargo{
        (|am.historiales|<n \implies |res| = |am.historiales|) \land \\
        (|am.historiales|\geq n \implies |res| = n) \land \\ 
        ((\forall j,k \in \Z)(\ j\in res \land k\in res \land) \imples j \neq k) \land \\
        ((\forall u \in \Z)  \ u\in res \implies u \in am.historiales) \land_L \\
        ((\forall i \in \Z) \ 0 \leq i < |res| - 1 \implicaLuego millasDisponibles(am.historiales[res[i]].transacciones)) \geq millasDisponibles(am.historiales[res[i+1]].transacciones) \land \\\lnot ((\exists u \in \Z) u \in am.historiales \land  u \notin res \land_L \\millasDisponibles(am.historiales[u].transacciones) > \\millasDisponibles(am.historiales[res[|res|-1]].transacciones))                       
    }
\end{proc}





\begin{proc}{resumenTransacciones}
{
\In am:AirMiles,
\In usuario:Usuario
}{ \\
\Tipo{\tupla{Transaccion, Transaccion, Transaccion, Transaccion}}
}
    \requiere{usuario \in am.historiales}
    \aseguraLargo{res_0 = \tupla{"Acumular",acumuladoPorViajes(am.historiales[usuario].transacciones)} \land \\
                res_1= \tupla{"Canjear",totalCanjeadas(am.historiales[usuario].transacciones)} \land \\
                res_2= \tupla{"Transferir",totalTransferidas(am.historiales[usuario].transacciones)} \land \\
                res_3= \tupla{"Recibir",totalRecibidas(am.historiales[usuario].transacciones)}}
\end{proc}


\begin{proc}{iniciarPromocion}
{
\Inout am:AirMiles,
\In nombrePromocion : NombrePromocion,
\In factor : Factor
}{}
    \requiere{am=AM_0 \land factor \geq 1 \land nombrePromocion \notin AM_0.promociones}
    \aseguraLargo{am.historiales = AM_0.historiales \land \\
        am.promociones = setKey(AM_0.promociones,nombrePromocion, \tupla{factor,False,0})       
        }
\end{proc}


\begin{proc}{iniciarPromocionConTope}
{
\Inout am:AirMiles,
\In nombrePromocion : NombrePromocion, \\
\In factor : Factor,
\In tope : montoTope
}{}
    \requiere{am=AM_0 \land factor \geq 1 \land nombrePromocion \notin AM_0.promociones}
    \aseguraLargo{am.historiales = AM_0.historiales \land \\
        am.promociones = setKey(AM_0.promociones,nombrePromocion, \tupla{factor,True,tope})
    }
\end{proc}


\begin{proc}{finalizarPromocion}
{
\Inout am:AirMiles,
\In nombrePromocion : NombrePromocion
}{}
    \requiere{am=AM_0\ \land  \ nombrePromocion \in AM_0.promociones\ \land \ nombrePromocion \neq "NoPromo"}
    \aseguraLargo{am.historiales = AM_0.historiales \land \\
        am.promociones = delKey(AM_0.promociones,nombrePromocion)
    }
\end{proc}


\begin{proc}{elQueTieneMasMillas}
{
\In am:AirMiles
}{
\Tipo{Usuario}
}
    \requiere{|am.historiales|>0}
    \aseguraLargo{
        res \in am.historiales \yLuego \\
        (\forall k \in \Z)(k \in am.historiales \implicaLuego \ millasTotales(am.historiales[k].transacciones)\\ \leq millasTotales(am.historiales[res].transacciones))
        }
\end{proc}

{

}

\end{proc}


\begin{proc}{paresTransferidoresExclusivos}
{
\In am:AirMiles
}{
\Tipo{\seq{tupla\tupla{Usuario,Usuario}}}
}
    \requiere{True}
    \aseguraLargo{
        (sinRepetidosNiAnagramas(res) \land \\
        (\forall i,j \in Z)(((i,j \in am.historiales \ \land i\neq j)\ \yLuego \\ sonParejaTransferidores(am.historiales,i,j))\implicaLuego (\tupla{i,j} \in res \lor \tupla{j,i} \in res )) \land \\ 
        (\forall k : Z)(0\leq k < |res| \implicaLuego ((res[k]_0 \in am.historiales \land res[k]_1\in am.historiales) \ \yLuego \\ sonParejaTransferidores(am.historiales,res[k]_0,res[k]_1)))
    }
\end{proc}



\pred{igualSalvo}{dict1: dicc<K, V>, dict2: dicc<K, V>, ignorar: \conj{Usuario}}{\\
    (\forall i : K) ((i \in dict1 \land i \notin ignorar) \implicaLuego (i \in dict2 \yLuego dict1[i] = dict2[i])) \land \\
    (\forall j : K) (j \in dict2 \land j \notin ignorar) \implicaLuego (j \in dict1 \yLuego dict1[j] = dict2[j]) 
}
\\

\pred{esPromocionSinTope}{am : AirMiles, promocion : NombrePromocion}{ \\ am.promociones[promocion]_1 = False}
\\
\\
\pred{esPromocionConTope}{am : AirMiles, promocion : NombrePromocion}{ \\ am.promociones[promocion]_1 = True}
\\
\\
\pred{esBronce}{trans:Transacciones}{millasTotales(trans) < 10.000}
\\
\\
\pred{esPlata}{trans:Transacciones}{10.000 \leq millasTotales(trans) < 50.000}
\\
\\
\pred{esOro}{trans:Transacciones}{50.000 \leq millasTotales(trans) < 1.000.000}
\\
\\
\pred{esPlatino}{trans:Transacciones}{millasTotales(trans) \geq 1.000.000}
\\
\\
\pred{historialActualizado}{am : AirMiles, AM_0 : AirMiles, u : Usuario, d : Distancia, cat : R, p : NombrePromocion}{ \\
am.historiales = setKey(AM_0.historiales, u,  \\ historialIndividualModificado(AM_0.historiales, u, "Acumular", d, factorPromo(AM_0, p), cat, p))
}
\\
\\
\pred{esUso}{accion:Accion}{ accion \in  \seq{"Transferir","Canjear"} }
\\
\\
\pred{ganaTotales}{accion:Accion}{ accion \in  \seq{"Recibir","Acumular","Referidor","Referido"} }
\\
\\
\pred{pierdeTotales}{accion:Accion}{ accion \in  \seq{"Transferir"} }
\\
\\
\pred{sonParejaTransferidores}{hist: Historiales, usuario1:Usuario,\\ usuario2:Usuario}{\\
(|hist[usuario1].transferencias| > 0) \land (|hist[usuario2].transferencias| > 0)  \land \\
(\forall i \in Z)(0 \leq i < |hist[usuario1].transferencias| \implicaLuego (hist[usuario1].transferencias[i]_2 = usuario2) \ \land 
(\forall j \in Z)(0 \leq j < |hist[usuario2].transferencias| \implicaLuego (hist[usuario2].transferencias[j]_2 = usuario1) \\
}
\\
\\
\pred {sinRepetidosNiAnagramas}{seqTuplUsu:\seq{\tupla{Usuario,Usuario}}}{\\
    (\forall i,j \in Z)(0 \leq i,j < |seqTuplUsu|\ \land \ i\neq j\ \implicaLuego ((seqTuplUsu[i]_0 \neq seqTuplUsu[j]_0 \land seqTuplUsu[i]_1 \neq seqTuplUsu[j]_1) \land \\(seqTuplUsu[i]_0 \neq seqTuplUsu[j]_1 \land \ seqTuplUsu[i]_1 \neq seqTuplUsu[j]_0)))
}
\\
\\
\pred{noSePasaDeTope}{am : AirMiles, usuario : Usuario, distancia : Distancia, \\ promocion : NombrePromocion}{ \\(esBronce(millasTotales(am.historiales[usuario].transacciones))\implies \\
                cantidadMillasAGanar(distancia,factorPromo(am.promociones,promocion),1) \leq \\ millasDisponiblesEnPromo(am.promociones,promocion))  \\ \land \\
                (esPlata(millasTotales(am.historiales[usuario].transacciones)) \implies \\
                cantidadMillasAGanar(distancia,factorPromo(am.promociones,promocion),1.5) \leq \\ millasDisponiblesEnPromo(am.promociones,promocion)) \\ \land \\
                (esOro(millasTotales(am.historiales[usuario].transacciones)) \implies \\
                cantidadMillasAGanar(distancia,factorPromo(am.promociones,promocion),2) \leq \\ millasDisponiblesEnPromo(am.promociones,promocion))}
\\
\\
\pred{alcanzaTope}{am : AirMiles, distancia : Distancia, categoria : R, promocion : NombrePromocion}{am.promociones[promocion]_2 = distancia * categoria * factorPromo(am, promocion)}
\\
\\
\pred{restaDelTope}{am : AirMiles, AM_0 : AirMiles, distancia : Distancia, categoria : R, promocion : NombrePromocion}{am.promociones = setKey(AM_0, promocion, \\ \tupla{AM_0.promociones[promocion]_0,AM_0.promociones[promocion]_1, \\ AM_0.promociones[promocion]_2-cantidadMillasAGanar(distancia,AM_0.promociones[promocion]_0,categoria))}}
\\
\\

\aux{millasTotales}{t: Transacciones}{\R}{ \\ \sum_{i=0}^{|t|-1} (IfThenElse(ganaTotales(t[i]_0), t[i]_1, 0) - ifThenElse(pierdeTotales(t[i]_0), t[i]_1, 0))}
\\
\\
\aux{cantidadMillasAGanar}{dist:Distancia,fact:Factor,cate:R}{Millas}{dist*fact*cate}
\\
\\
\aux{factorPromo}{am : AirMiles, promocion : NombrePromocion}{Factor}{am.promociones[promocion]_0}
\\
\\
\aux{millasDisponiblesEnPromo}{promos:Promociones, promo:NombrePromocion}{Millas}{promos[promocion]_2}
\\
\\
\aux{historialIndividualModificado}{hist:Historiales,u:Usuario,accion:Accion,d:Distancia, f:Factor,cat:R,p:NombrePromocion}{Historial}{
            \\struct<transacciones: hist[u].transacciones++\seq{accion,d*f*cat}, \\ vuelos: hist[u].vuelos ++ \seq{\tupla{d,d*f*cat,p,f}},\\ transferencias: hist[u].transferencias>
}
\\
\\
\aux{millasDisponibles}{trans: Transacciones}{\R}{$\sum_{i=0}^{|trans|-1}$ ifThenElse(esUso(trans[i]_0)), -1*trans[i]_1, trans[i]_1)} 
\\
\\
\aux{acumuladoPorViajes}{trans:Transacciones}{Millas}{ \\ $\sum_{i=0}^{|trans|-1}$ ifThenElse(trans[i]_0 ="Acumular",trans[i]_1,0)}
\\
\\
\aux{totalCanjeadas}{trans:Transacciones}{Millas}{$\sum_{i=0}^{|trans|-1}$ \\ifThenElse(trans[i]_0 ="Canjear",trans[i]_1,0)}
\\
\\
\aux{totalGastadas}{trans:Transacciones}{Millas}{$\sum_{i=0}^{|trans|-1}$ \\ifThenElse(trans[i]_0 \in \seq{"Canjear","Transferir"},trans[i]_1,0)}
\\
\\
\aux{totalTransferidas}{trans:Transacciones}{Millas}{$\sum_{i=0}^{|trans|-1}$ \\ifThenElse(trans[i]_0 = "Transferir",trans[i]_1,0)}
\\
\\
\aux{totalRecibidas}{trans:Transacciones}{Millas}{$\sum_{i=0}^{|trans|-1}$ \\ifThenElse(trans[i]_0 = "Recibir",trans[i]_1,0)}
\\
