\begin{proc}{acumularMillas}
{

\In usuario:Usuario,
\In distancia: Distancia,
\In promocion: NombrePromocion
}{}
    \requiere{am = AM_0 \land usuario \in AM_0.historiales \land promocion \in AM_0.promociones \land \\
                (promocionSinTope(AM_0.promociones, promocion)\lor \\(esBronce(millasTotales(AM_0.historiales[usuario].transacciones))\implies \\
                cantidadMillasAGanar(distancia,factorPromo(AM_0.promociones,promocion),1) \leq \\ millasDisponiblesEnPromo(AM_0.promociones,promocion)) \land \\ 
                esPlata(millasTotales(AM_0.historiales[usuario].transacciones) \implies \\
                cantidadMillasAGanar(distancia,factorPromo(AM_0.promociones,promocion),1.5) \leq \\ millasDisponiblesEnPromo(AM_0.promociones,promocion))\land \\
                esOro(millasTotales(AM_0.historiales[usuario].transacciones)) \implies \\
                cantidadMillasAGanar(distancia,factorPromo(AM_0.promociones,promocion),2) \leq \\ millasDisponiblesEnPromo(AM_0.promociones,promocion) )
                )
    }
    \aseguraLargo{ ((promocionSinTope(AM_0.promociones, promocion) \lor
                    \\esPlatino(AM_0.historiales[usuario].transacciones)) \implies am.promociones= AM_0.promociones\land \\
                    ¬promocionSinTope(AM_0.promociones, promocion) \implies \\ ( (millasAcumuladas() = millasDisponiblesEnPromo(AM_0.promociones,promocion) \implies \\ am.promociones = delKey(AM_0.promociones,promocion) ) \lor \\                    (esTopeModificadoBronce(am.promociones,AM_0.promociones,promocion,AM_0.historiales,usuario,distancia) \lor \\       esTopeModificadoPlata(am.promociones,AM_0.promociones,promocion,AM_0.historiales,usuario,distancia) \lor \\            esTopeModificadoOro(am.promociones,AM_0.promociones,promocion,AM_0.historiales,usuario,distancia) ) \land\\
                    esBronce(AM_0.historial[usuario].historialTransaccion)) \implies \\ esHistorialActualizadoBronce(am.historiales, AM_0.historiales,usuario, distancia,\\ factorPromo(AM_0.promociones,promocion),promocion) \land \\
                    esPlata(AM_0.historiales[usuario].transacciones)) \implies \\ esHistorialActualizadoPlata(am.historiales, AM_0.historiales,usuario, distancia,\\ factorPromo(AM_0.promociones,promocion),promocion) \land \\
                    esOro(AM_0.historiales[usuario].transacciones)) \implies \\ esHistorialActualizadoOro(am.historiales, AM_0.historiales,usuario, distancia,\\ factorPromo(AM_0.promociones,promocion),promocion) \land\\
                    esPlatino(AM_0.historiales[usuario].transacciones)) \implies \\ esHistorialActualizadoPlatino(am.historiales, AM_0.historiales,usuario, distancia,\\ factorPromo(AM_0.promociones,promocion),promocion) )
                    }                  

\end{proc}


\begin{proc}{canjearMillas}
{
\Inout am:AirMiles,
\In usuario:Usuario,
\In millas:Millas
}{}
    \requiere{am=AM_0\  \land \ usuario\in AM_0.historiales  \land \\ millas \leq millasDisponibles(millasSinCanjeo(AM_0.historiales[usuario].transacciones))}
    \aseguraLargo{
        am.promociones = AM_0.promociones \land \\
        igualSalvo(am,AM_0,\tupla{usuario}) \land \\
        am.historial[usuario].transacciones = AM_0.historial[usuario].transacciones ++ \seq{\tupla{"Canjear",millas}} \land \\
        am.historial[usuario].viajes = AM_0.historial[usuario].viajes \land \\
        am.historial[usuario].transferencias = AM_0.historial[usuario].transferencias        
                }
\end{proc}


\begin{proc}{transferirMillas}
{
\Inout am:AirMiles,
\In dador : Usuario,
\In receptor : Usuario,\\
\In millas : Millas
}{}    
    \requiere{am=AM_0 \land dador \in AM_0.historiales \land \\receptor \in AM_0.historiales \land \\ dador \neq receptor \land \\ millasDisponibles(millasSinCanjeo(AM_0.historiales[dador].transacciones)) \geq millas}
    \aseguraLargo{am.promociones = AM_0.promociones \land \\
                igualesSalvo(am,AM_0,\seq{dador,receptor} \land \\
                am.historial[dador].vuelos = AM_0.historial[dador].vuelos \land \\
                am.historial[dador].transacciones = AM_0.historial[dador].transacciones ++ \seq{\tupla{"Transferir",millas}} \land \\
                am.historial[dador].transferencias = AM_0.historial[dador].transferencias ++ \seq{receptor} \land \\
                am.historial[receptor].vuelos = AM_0.historial[receptor].vuelos \land \\
                am.historial[receptor].transacciones = AM_0.historial[receptor].transacciones ++ \seq{\tupla{"Transferir",millas}} \land \\
                am.historial[receptor].transferencias = AM_0.historial[receptor].transferencias ++ \seq{dador}
                
                    }  
\end{proc}

\begin{proc}{consultarCategoria}{
\In am:AirMiles,
\In usuario:Usuario
}{
\Tipo{Categoria}
}
    \requiere{usuario \in am.historial}
    \aseguraLargo{(res = "Bronces" \iff totalDeMillas(am.historial[usuario].transacciones) < 10.000) \land \\
            (res = "Plata" \iff (totalDeMillas(am.historial[usuario].transacciones) \geq 10.000 \land totalDeMillas(am.historial[usuario].transacciones) < 50.000) \land \\
            (res = "Oro" \iff (totalDeMillas(am.historial[usuario].transacciones) \geq 50.000 \land totalDeMillas(am.historial[usuario].transacciones) < 1.000.000) \land \\
            (res = "Platino" \iff totalDeMillas(am.historial[usuario].transacciones) \geq 1.000.000) \\
            }
\end{proc}

\begin{proc}{rankingMillas}
{
\In am:AirMiles,
\In n:$\mathbb{Z}$
}{
\Tipo{\seq{Usuario}}
}
    \requiere{n > 0}
    \aseguraLargo{
        (|am.historiales|<n \implies |res| = |am.historiales|) \land \\
        %Aca no deberia ser (|am.historiales|<n \implies res = claves(am.historiales)). Porque lo que queremos es que nos devuelva todos lo usuarios del sistema
        (|am.historiales|\geq n \implies |res| = n) \land \\ ((\forall u \in \Z)  \ u\in res \implies u \in am.historiales) \land_L \\
        %Esta bien ¿\ u\in res? siempre lo vi como 0 <= i < |res| ->_l res[i] \in am.historiales
        ((\forall i \in \Z) \ 0 \leq i < |res| - 1 \implicaLuego millasDisponibles(millasSinCanjeo(am.historiales[res[i]].transacciones)) \geq millasDisponibles(millasSinCanjeo(am.historiales[res[i+1]].transacciones)) \land \\\lnot ((\exists u \in \Z) u \in am.historiales \land  u \notin res \land \\millasDisponibles(millasSinCanjeo(am.historiales[u].transacciones)) > \\millasDisponibles(millasSinCanjeo(am.historiales[res[|res|-1]].transacciones)))                       
    }
\end{proc}
